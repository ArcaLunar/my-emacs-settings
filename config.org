** Font

Monitor specific settings

#+begin_src emacs-lisp
(let ((mon-name (alist-get 'name (frame-monitor-attributes)))
      (font-size 14)
      (monospace-font "Lilex")
      (writing-font "Anthropic Serif Text"))
  (when (equal mon-name "eDP-1")
    (setq font-size 20))

  (setq doom-font
        (font-spec
         :family monospace-font
         :size font-size))
  (setq doom-variable-pitch-font
        (font-spec
         :family writing-font
         :size font-size)))

;; CJK fonts
(defun my/setup-cjk-fonts ()
  (dolist (script '(han kana cjk-misc bopomofo))
    (set-fontset-font nil script
                      (font-spec
                       :family "LXGW WenKai Mono Screen")
                      nil 'prepend)))
(if (display-graphic-p)
    (progn
      (add-hook 'after-setting-font-hook #'my/setup-cjk-fonts)
      (my/setup-cjk-fonts)))
#+end_src

* Org Management

#+begin_src emacs-lisp
(defun my/org-export-to-tmp-html ()
  "Export current org buffer to a temp HTML file in /tmp and open it in the default browser."
  (interactive)
  (require 'ox-html)
  (let ((tmpfile (make-temp-file "org-export-" nil ".html")))
    (org-export-to-file 'html tmpfile)
    (browse-url (concat "file://" tmpfile))
    (message "Exported to %s" tmpfile)))
#+end_src
